Create the Dockerfile

        
Use the generator above, or write it by hand. Save it as Dockerfile with no extension in your project root.

        

          
          
FROM node:20-alpine AS base
WORKDIR /app
COPY package*.json ./
RUN npm ci --omit=dev
COPY . .
RUN addgroup -S appgroup && adduser -S appuser -G appgroup
USER appuser
EXPOSE 3000
CMD ["node", "server.js"]

        

      


      
      

        
Add a .dockerignore

        
Keeps the build context small and stops secrets from leaking into the image.

        

          
          
node_modules
npm-debug.log
.git
.env
Dockerfile
*.md
dist
coverage

        

      


      
      

        
Build the image

        

          
          
docker build -t myapp:1.0 .
docker images | grep myapp

        

      


      
      

        
Run it locally

        

          
          
docker run -d -p 3000:3000 --name myapp-container myapp:1.0
docker logs -f myapp-container
curl http://localhost:3000

        

      


      
      

        
Push to a registry

        

          
          
docker tag myapp:1.0 yourusername/myapp:1.0
docker login
docker push yourusername/myapp:1.0

        

      


      
      

        
Deploy on the server

        

          
          
docker pull yourusername/myapp:1.0
docker run -d \
  --name myapp \
  --restart unless-stopped \
  -p 80:3000 \
  --env-file .env.production \
  yourusername/myapp:1.0
